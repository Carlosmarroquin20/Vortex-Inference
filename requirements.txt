# Core inference framework
# llama-cpp-python provides Python bindings to llama.cpp's C++ inference engine.
# Version pinned to ensure reproducible builds — llama.cpp undergoes frequent
# breaking API changes between minor versions due to active GGUF format evolution.
llama-cpp-python==0.3.36

# HTTP framework
# FastAPI is chosen over Flask for native async support — critical for serving
# concurrent health checks and metrics scrapes without blocking the inference thread.
fastapi==0.143.0
uvicorn[standard]==0.54.0

# Starlette is pinned explicitly even though FastAPI pulls it in transitively.
# Older FastAPI releases capped it below 0.38, which kept several published
# request-parsing CVEs in the image. Pinning it here makes the security-relevant
# version visible in review and to dependency scanners instead of leaving it to
# the resolver.
starlette==1.7.0

# Configuration management via Pydantic v2
pydantic==2.14.1
pydantic-settings==2.15.0

# Prometheus instrumentation
# prometheus-client is the reference implementation — no dependencies, stable API.
prometheus-client==0.26.0

# HTTP client for remote inference backends (app/backends/openai_compatible.py).
# httpx2 rather than httpx: it is the client Starlette 1.x and the OpenAI SDK
# build on, so the image and the test suite share one HTTP stack.
httpx2==2.13.1

# Official SDK for the anthropic backend (app/backends/anthropic_api.py). The
# 1.x line runs on httpx2, so it shares the client pinned above.
anthropic==1.13.0
